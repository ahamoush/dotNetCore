namespace ProjectManagementSystem.Application.DTOs.Auth;

public class AuthResult
{
    public bool Succeeded { get; init; }

    public AuthResponseDto? Data { get; init; }

    public string[] Errors { get; init; } = [];

    public static AuthResult Success(AuthResponseDto data)
        => new() { Succeeded = true, Data = data };

    public static AuthResult Failure(params string[] errors)
        => new() { Succeeded = false, Errors = errors };
}

خصائص وطرق

العضوالنوعالمعنى

DataAuthResponseDto? (init)البيانات عند النجاح؛ قد تكون null.
Errorsstring[] (init)رسائل الخطأ؛ افتراضيًا مصفوفة فارغة.

نمط Result يمكّن الخدمة من إرجاع سبب الفشل دون رمي استثناء لكل حالة متوقعة.

ملخص

نوع مغلق للإنشاء (init) مع مصنعين Success وFailure.
