FluentValidation لمعاملات ProjectQueryDto.

الكود في المشروع

using FluentValidation;
using ProjectManagementSystem.Application.DTOs;

namespace ProjectManagementSystem.Application.Validators;

public sealed class ProjectQueryDtoValidator : AbstractValidator<ProjectQueryDto>
{
    private static readonly string[] AllowedSortFields = ["name", "createdat", "duedate"];

    public ProjectQueryDtoValidator()
    {
        RuleFor(x => x.Search)
            .MaximumLength(200);

        RuleFor(x => x.PageNumber)
            .GreaterThanOrEqualTo(1);

        RuleFor(x => x.PageSize)
            .InclusiveBetween(1, 100);

        RuleFor(x => x.SortBy)
            .Must(sortBy => AllowedSortFields.Contains(sortBy.Trim(), StringComparer.OrdinalIgnoreCase))
            .When(x => !string.IsNullOrWhiteSpace(x.SortBy))
            .WithMessage("SortBy must be one of: Name, CreatedAt, DueDate.");
    }
}

الحقول المسموحة للفرز

المصفوفة الثابتة AllowedSortFields: name، createdat، duedate (بحروف صغيرة في المصدر).

الحقلالقواعد
SearchMaximumLength(200)
PageNumberGreaterThanOrEqualTo(1)
PageSizeInclusiveBetween(1, 100)
SortByMust: بعد Trim يجب أن يكون ضمن AllowedSortFields (مقارنة غير حساسة لحالة الأحرف) — When إذا لم يكن SortBy فارغاً أو مسافات فقط؛ رسالة: «SortBy must be one of: Name, CreatedAt, DueDate.»
ملاحظة: الرسالة تذكر DueDate بين الحقول المسموحة، بينما كائن الاستعلام للمشاريع لا يحتوي حقل استحقاق في ProjectQueryDto نفسه — القيمة duedate موجودة فقط في قائمة المدقق كما في الكود.
لا قواعد صريحة على Status أو Descending في هذا الملف.
